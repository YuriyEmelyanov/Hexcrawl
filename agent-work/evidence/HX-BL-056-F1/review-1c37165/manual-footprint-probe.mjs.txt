// Save as test/review-lake-boundaries-1c.mjs and run from the repository root with Node 24.
import fs from 'node:fs';
import assert from 'node:assert/strict';
import {createGenerationHarness} from './helpers/generation-harness.mjs';
import {hasFullDrySupport,hasFullOneMileSupport,smallLakeNeighbours,parseSmallLakes,addManualSmallLake} from '../src/smallLakes.ts';
import {fineCellKey} from '../src/oneMile.ts';

const document=JSON.parse(fs.readFileSync(new URL('./fixtures/e6-water-editor.json',import.meta.url)));
const load=(h,input)=>h.render().handleImportJson({target:{files:[{text:JSON.stringify(input)}],value:'file'}});
const serial=value=>JSON.parse(JSON.stringify(value));
const stable=app=>{const out=serial(app.createSaveData());delete out.savedAt;return out;};
const h=createGenerationHarness(42);
h.render();
const alerts=[];
h.setWindow({alert:message=>alerts.push(message),matchMedia:()=>({matches:false})});
h.injectFunction('readTextFile',async file=>file.text);
await load(h,document);
assert.equal(h.render().switchFineStep(1),true);
const grid=h.render().fineCells;
const key=fineCellKey;
const usable=new Set(grid.filter(hasFullDrySupport).map(x=>key(x.cell)));
const pick=cell=>{const a=h.render();return a.pickFine(cell,false,a.getDocumentRevision());};
const before=stable(h.render());
let expanded;
for(const row of grid.filter(hasFullDrySupport).slice(0,20)){
  const cells=[0,1,2,3].map(offset=>({q:row.cell.q+offset,r:row.cell.r}));
  if(!cells.every(c=>usable.has(key(c))))continue;
  assert.ok(pick(cells[0]));
  h.render().startSmallLake();
  cells.slice(1).forEach(cell=>assert.ok(pick(cell)));
  h.render().commitSmallLake();
  assert.equal(h.render().editError,null);
  expanded=stable(h.render());
  assert.deepEqual(expanded.map.physicalGrid.smallLakes.lakes[0].cells,cells);
  break;
}
assert.ok(expanded);
const used=expanded.map.physicalGrid.smallLakes.lakes[0];
const covered=new Set([...used.cells,...used.shoreCells].map(key));
const actualPartial=grid.find(x=>!hasFullOneMileSupport(x)&&!covered.has(key(x.cell))&&smallLakeNeighbours(x.cell).some(c=>usable.has(key(c))&&!covered.has(key(c))));
assert.ok(actualPartial);
const shoreStart=smallLakeNeighbours(actualPartial.cell).find(c=>usable.has(key(c))&&!covered.has(key(c)));
assert.ok(pick(shoreStart));h.render().startSmallLake();
const revision=h.render().getDocumentRevision();
assert.ok(pick(actualPartial.cell));
const afterSelection=stable(h.render());
const stacks=serial([h.render().undoHistory,h.render().redoHistory]);
h.render().commitSmallLake();
assert.deepEqual(stable(h.render()),afterSelection);
assert.deepEqual(serial([h.render().undoHistory,h.render().redoHistory]),stacks);
assert.ok(h.render().editError);
assert.throws(()=>addManualSmallLake(expanded.map.physicalGrid.smallLakes,[shoreStart,actualPartial.cell],grid),/no full dry support/);
assert.equal(h.render().getDocumentRevision(),revision);
const duplicate=structuredClone(expanded);
duplicate.map.physicalGrid.smallLakes.nextId=3;
duplicate.map.physicalGrid.smallLakes.lakes.push({...used});
assert.throws(()=>parseSmallLakes(duplicate.map.physicalGrid.smallLakes),/Invalid small lake/);
await load(h,duplicate);
assert.equal(alerts.length,1);
assert.deepEqual(stable(h.render()),afterSelection);
console.log(JSON.stringify({four:used.cells,partial:actualPartial.cell,adjacent:shoreStart,duplicate_refused:true,undo_preserved:true,beforeHashable:before.version}));
