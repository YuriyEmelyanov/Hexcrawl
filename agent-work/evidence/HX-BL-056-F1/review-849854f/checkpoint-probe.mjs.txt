// Run with Node 24 as test/reviewer-checkpoint-probe.mjs in either checkout. Uses the unmodified generation harness.
import fs from 'node:fs';
import assert from 'node:assert/strict';
import {createGenerationHarness} from './helpers/generation-harness.mjs';
const fixture=JSON.parse(fs.readFileSync(new URL('./fixtures/e6-water-editor.json',import.meta.url)));
const h=createGenerationHarness(42);h.render();
const alerts=[],confirms=[];
h.setWindow({alert:x=>alerts.push(x),confirm:m=>(confirms.push(m),true),matchMedia:()=>({matches:false})});
h.injectFunction('readTextFile',async file=>file.text);
const load=data=>h.render().handleImportJson({target:{files:[{text:JSON.stringify(data)}],value:'file'}});
const snap=()=>JSON.parse(JSON.stringify(h.render().createSaveData()));
const gate=()=>assert.deepEqual(alerts,[]);
const pick=cell=>{const app=h.render();assert.ok(app.pickFine(cell,false,app.getDocumentRevision()));};
const switchTo=n=>assert.ok(h.render().switchFineStep(n));
const manual=cell=>{pick(cell);h.render().startSmallLake();h.render().commitSmallLake();assert.equal(h.render().editError,null);};
 await load(fixture);gate();switchTo(1);manual({q:-21,r:-11});switchTo(6);
 const modern=snap(),legacy=structuredClone(modern);delete legacy.map.physicalGrid.smallLakes.lakes[0].shoreCells;
 for(const [version,data] of [['modern',modern],['legacy',legacy]]){
  await load(data);gate();
  const before=snap();assert.equal(before.map.physicalGrid.smallLakes.nextId,2);const generated=h.render().safelyAddRegionToMap({q:3,r:3});
  assert.ok(generated.success);const after=snap();h.render().deleteLastRegion();const rollback=snap();
  const again=h.render().safelyAddRegionToMap({q:3,r:3});assert.ok(again.success);const second=snap();
  const initial=after.map.physicalGrid.smallLakes.lakes.slice(1),repeated=second.map.physicalGrid.smallLakes.lakes.slice(1);
  const geog=lakes=>lakes.map(({id,...rest})=>rest);
  const replay={firstIds:initial.map(l=>l.id),secondIds:repeated.map(l=>l.id),sameGeography:JSON.stringify(geog(initial))===JSON.stringify(geog(repeated))};
  assert.deepEqual(alerts,[]);assert.equal(rollback.map.regions.length,1);assert.equal(rollback.map.physicalGrid.smallLakes.lakes.length,1);
  console.log(JSON.stringify({version,replay,pre:before.map.physicalGrid.smallLakes,generatedCount:after.map.physicalGrid.smallLakes.lakes.length,next:after.map.physicalGrid.smallLakes.nextId,post:rollback.map.physicalGrid.smallLakes,regions:rollback.map.regions.length,confirm:confirms.at(-1),error:h.render().editError}));
 }
