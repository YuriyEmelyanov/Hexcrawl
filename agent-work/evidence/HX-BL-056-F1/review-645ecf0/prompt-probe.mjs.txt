// Run with Node 24 as test/reviewer-checkpoint-probe.mjs in either checkout. Uses the unmodified generation harness.
import fs from 'node:fs';
import assert from 'node:assert/strict';
import {createGenerationHarness} from './helpers/generation-harness.mjs';
const fixture=JSON.parse(fs.readFileSync(new URL('./fixtures/e6-water-editor.json',import.meta.url)));
const h=createGenerationHarness(42);h.render();
const alerts=[],confirms=[];
h.setWindow({alert:x=>alerts.push(x),confirm:m=>(confirms.push(m),true),matchMedia:()=>({matches:false})});
h.injectFunction('readTextFile',async file=>file.text);
const load=data=>h.render().handleImportJson({target:{files:[{text:JSON.stringify(data)}],value:'file'}});
const snap=()=>JSON.parse(JSON.stringify(h.render().createSaveData()));
const gate=()=>assert.deepEqual(alerts,[]);
const pick=cell=>{const app=h.render();assert.ok(app.pickFine(cell,false,app.getDocumentRevision()));};
const switchTo=n=>assert.ok(h.render().switchFineStep(n));
const manual=cell=>{pick(cell);h.render().startSmallLake();h.render().commitSmallLake();assert.equal(h.render().editError,null);};
 await load(fixture);gate();switchTo(1);manual({q:-21,r:-11});switchTo(6);
 const modern=snap(),legacy=structuredClone(modern);delete legacy.map.physicalGrid.smallLakes.lakes[0].shoreCells;

for(const withNote of [false,true]){
 await load(legacy);gate();assert.ok(h.render().safelyAddRegionToMap({q:3,r:3}).success);
 const c=h.render().regions.at(-1).hexes[0],key=c.q+','+c.r;
 if(withNote){h.render().setSelectedHex(c);h.render().changeNotesText(key,'new region');h.render().finishNotesGroup();assert.equal(snap().map.gmNotesByHexKey[key],'new region');}
 const start=confirms.length,depth=h.render().undoHistory.length,before=snap();
 h.render().deleteLastRegion();assert.equal(snap().map.regions.length,1);
 console.log(JSON.stringify({withNote,key,prompts:confirms.slice(start),transactions:h.render().undoHistory.length-depth}));
}
