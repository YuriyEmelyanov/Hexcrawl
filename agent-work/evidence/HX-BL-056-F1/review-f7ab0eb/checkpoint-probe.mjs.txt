import assert from 'node:assert/strict';
import fs from 'node:fs';
import {createGenerationHarness} from './helpers/generation-harness.mjs';

const h=createGenerationHarness(42);
const alerts=[], prompts=[];
h.setWindow({alert:v=>alerts.push(v),confirm:v=>(prompts.push(v),true),matchMedia:()=>({matches:false})});
h.injectFunction('readTextFile',async file=>file.text);
const fixture=JSON.parse(fs.readFileSync(new URL('./fixtures/e6-water-editor.json',import.meta.url)));
const load=data=>h.render().handleImportJson({target:{files:[{text:JSON.stringify(data)}],value:'file'}});
const snap=()=>{const value=JSON.parse(JSON.stringify(h.render().createSaveData()));delete value.savedAt;return value;};
const timeline=()=>JSON.parse(JSON.stringify([h.render().undoHistory,h.render().redoHistory]));
fixture.map.roads=[{id:1,regionId:1,segments:[{from:{q:0,r:-1},to:{q:1,r:-1},kind:'trail'}]}];fixture.map.nextRoadId=2;
await load(fixture);
assert.deepEqual(alerts,[]);
const road=h.render().roads[0];
console.log('initial', JSON.stringify({road,regions:h.render().regions.length}));
h.render().addFallbackTractToMap(h.render().candidateHexes[0]);
console.log('after generation', JSON.stringify({regions:h.render().regions.length,history:h.render().history.length,error:h.render().generationError}));
assert.equal(h.render().regions.length,2);
const removed=h.render().editRoadDocument({type:"remove",roadId:road.id});console.log("remove",removed,h.render().editError);assert.equal(removed,true);
assert.equal(h.render().switchFineStep(1),true);
const {pickFine,getDocumentRevision}=h.render();
assert.equal(pickFine({q:3,r:-6},false,getDocumentRevision()),true);
h.render().startSmallLake();
h.render().commitSmallLake();
assert.equal(h.render().editError,null);
assert.equal(h.render().switchFineStep(6),true);
const before=snap(),historyLength=h.render().undoHistory.length;
await load(before);
assert.deepEqual(alerts,[]);
h.render().undoMap();
assert.equal(h.render().undoHistory.length,historyLength);
h.render().deleteLastRegion();
const after=snap();
console.log('delete result',JSON.stringify({regionCount:after.map.regions.length,lakes:after.map.physicalGrid?.smallLakes,roads:after.map.roads,historyDelta:h.render().undoHistory.length-historyLength,prompts,editError:h.render().editError}));
assert.equal(after.map.regions.length,1);
assert.equal(after.map.roads.length,1);
assert.equal(after.map.physicalGrid.smallLakes.lakes.length,1);
h.render().undoMap();
assert.deepEqual(snap(),before);
h.render().redoMap();
assert.deepEqual(snap(),after);
const historyAtImport=timeline();
await load(after);
assert.deepEqual(snap(),after);
assert.deepEqual(timeline(),historyAtImport);
console.log('reimport alerts',JSON.stringify(alerts));
assert.match(alerts.at(-1),/Берег малого озера пересекает дорогу или тропу №1/);
